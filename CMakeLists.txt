cmake_minimum_required(VERSION 3.12)
project(lord_sumi_embedded_cpp LANGUAGES CXX)
enable_testing()
add_executable(cpp_tests cpp/tests.cpp)
target_include_directories(cpp_tests PRIVATE cpp)
set_target_properties(cpp_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cpp_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME cpp_tests COMMAND cpp_tests)
